  let selectedName = null;
  let selectedDesc = null;
  let matchedCount = 0;
  
  let currentQuizIndex = 0;
  let userScore = 0;
  let wrongConcepts = [];

  function updateProgress(percent) {
    document.getElementById('progressFill').style.width = percent + '%';
  }

  function switchScreen(screenId) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById(screenId).classList.add('active');
  }

  // --- Matching Game Logic ---
  function startMatchingGame() {
    switchScreen('screenMatching');
    updateProgress(20);
    renderMatchingItems();
  }

  function shuffle(array) {
    return array.sort(() => Math.random() - 0.5);
  }

  let shuffledNames = shuffle([...matchData]);
  let shuffledDescs = shuffle([...matchData]);

  function renderMatchingItems() {
    const nameContainer = document.getElementById('matchNames');
    const descContainer = document.getElementById('matchDescs');
    nameContainer.innerHTML = '';
    descContainer.innerHTML = '';

    shuffledNames.forEach(item => {
      const div = document.createElement('div');
      div.className = `match-item ${item.matched ? 'matched' : ''}`;
      div.innerText = item.name;
      div.dataset.id = item.id;
      if (!item.matched) {
        div.onclick = () => selectName(item.id, div);
      }
      nameContainer.appendChild(div);
    });

    shuffledDescs.forEach(item => {
      const div = document.createElement('div');
      div.className = `match-item ${item.matched ? 'matched' : ''}`;
      div.innerText = item.desc;
      div.dataset.id = item.id;
      if (!item.matched) {
        div.onclick = () => selectDesc(item.id, div);
      }
      descContainer.appendChild(div);
    });
  }

  function selectName(id, el) {
    if (el.classList.contains('matched')) return;
    document.querySelectorAll('#matchNames .match-item').forEach(i => i.classList.remove('selected'));
    el.classList.add('selected');
    selectedName = id;
    checkMatch();
  }

  function selectDesc(id, el) {
    if (el.classList.contains('matched')) return;
    document.querySelectorAll('#matchDescs .match-item').forEach(i => i.classList.remove('selected'));
    el.classList.add('selected');
    selectedDesc = id;
    checkMatch();
  }

  function checkMatch() {
    if (selectedName !== null && selectedDesc !== null) {
      if (selectedName === selectedDesc) {
        // Correct match
        matchData.find(i => i.id === selectedName).matched = true;
        shuffledNames.find(i => i.id === selectedName).matched = true;
        shuffledDescs.find(i => i.id === selectedDesc).matched = true;
        matchedCount++;

        selectedName = null;
        selectedDesc = null;
        renderMatchingItems();

        if (matchedCount === matchData.length) {
          document.getElementById('matchNextBtn').style.display = 'block';
        }
      } else {
        // Incorrect match feedback animation or reset
        setTimeout(() => {
          document.querySelectorAll('.match-item').forEach(i => i.classList.remove('selected'));
          selectedName = null;
          selectedDesc = null;
        }, 400);
      }
    }
  }

  // --- Quiz Logic ---
  function startQuizStage() {
    switchScreen('screenQuiz');
    updateProgress(40);
    renderQuiz();
  }

  function renderQuiz() {
    const q = quizData[currentQuizIndex];
    const container = document.getElementById('quizContainer');
    
    let html = `
      <div class="quiz-question">${q.question}</div>
      <div class="options-list">
    `;
    
    q.options.forEach((opt, idx) => {
      html += `<button class="option-btn" onclick="handleAnswer(${idx})">${String.fromCharCode(65 + idx)}. ${opt}</button>`;
    });
    
    html += `</div><div id="quizExplanationArea"></div>`;
    container.innerHTML = html;
  }

  function handleAnswer(selectedIndex) {
    const q = quizData[currentQuizIndex];
    const optionBtns = document.querySelectorAll('.option-btn');
    
    // Disable all options after selection
    optionBtns.forEach((btn, idx) => {
      btn.disabled = true;
      if (idx === q.correct) {
        btn.classList.add('correct');
      } else if (idx === selectedIndex) {
        btn.classList.add('incorrect');
      }
    });

    const isCorrect = (selectedIndex === q.correct);
    if (isCorrect) {
      userScore += 25;
    } else {
      wrongConcepts.push(q.conceptTag);
    }

    // Render Explanation Table
    const expArea = document.getElementById('quizExplanationArea');
    expArea.innerHTML = `
      <div class="explanation-box">
        <h4><i class="fa-solid fa-circle-info" style="color:${isCorrect ? '#22c55e' : '#ef4444'}"></i> ${isCorrect ? '정답입니다! 🎉' : '아쉽네요, 오개념을 바로 잡아볼까요?'}</h4>
        <table class="exp-table">
          <tr><th>선지 구성</th><td>${q.explanation.선지}</td></tr>
          <tr><th>정오 판정</th><td>${q.explanation.정오}</td></tr>
          <tr><th>내 생각</th><td>${q.explanation.생각}</td></tr>
          <tr><th>바로잡기</th><td><b>${q.explanation.바로잡기}</b></td></tr>
        </table>
